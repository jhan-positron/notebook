// Hiccup probe (scratch): the fake device thread's wait pattern on one CPU, to see whether the core stalls
// when it wakes and spins. Each cycle: nanosleep until (next - 100 us) [like FakeDevice::run], then spin on
// CLOCK_MONOTONIC until `next`, then post: lateness = now - next. Inside the spin, the largest gap between two
// clock reads is the stall. Cycle period is 1.2-2.6 ms (the churn trace's pass periods), varied per cycle.
// Mode "sleep": as above. Mode "spin": no sleep at all (the core never idles), the same accounting.
// Usage: hiccup CPU CYCLES MODE
#define _GNU_SOURCE
#include <sched.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

static int64_t now_ns(void) {
  struct timespec t;
  clock_gettime(CLOCK_MONOTONIC, &t);
  return (int64_t)t.tv_sec * 1000000000LL + t.tv_nsec;
}

static int cmp(const void* a, const void* b) {
  int64_t x = *(const int64_t*)a, y = *(const int64_t*)b;
  return (x > y) - (x < y);
}

static void stats(const char* name, int64_t* v, int n) {
  qsort(v, n, sizeof *v, cmp);
  double sum = 0;
  int over2 = 0, over50 = 0, over90 = 0;
  for (int i = 0; i < n; ++i) {
    sum += v[i];
    over2 += v[i] > 2500;
    over50 += v[i] > 50000;
    over90 += v[i] > 90000;
  }
  printf("%-14s n=%d mean=%.2f p50=%.2f p90=%.2f p99=%.2f max=%.1f us; >2.5us %d >50us %d >90us %d\n", name, n,
         sum / n / 1e3, v[n / 2] / 1e3, v[n * 9 / 10] / 1e3, v[n * 99 / 100] / 1e3, v[n - 1] / 1e3, over2, over50,
         over90);
}

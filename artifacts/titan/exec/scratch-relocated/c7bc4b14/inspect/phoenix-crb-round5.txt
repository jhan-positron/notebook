def placement(core: int, job: Iterable[int]) -> tuple[list[int], JsonObject]:
    """Where a run pinned to `core` places its threads, and the isolation that its summary
    records: the pass thread on `core`, which it pins itself to, and the runs' other threads on
    the job's CPUs `job` less the core's SMT siblings, which they leave alone, so that the pass
    thread has its core to itself; and that isolation, in the form that the demo's D17 lane
    records its own: the host, its boot and the LSF job; the core, its siblings and the CPUs of
    the run, of the job and of the host; and the kernel's isolated CPUs (isolcpus), and whether
    the core is among them. The latency judge reads U5 only on a run whose summary records it
    (spec §11.9)."""
    cpus = sorted(job)
    siblings = {core} | _cpus(CPU_SYSFS / f"cpu{core}" / "topology" / "thread_siblings_list")
    others = [c for c in cpus if c not in siblings]
    if not others:
        raise InfraError(f"the job's CPUs {cpus} leave the run no core beside the pass thread's, "
                         f"{core}")
    try:
        boot: str | None = BOOT_ID.read_text(encoding="utf-8").strip() or None
    except OSError:
        boot = None
    isolated = sorted(_cpus(CPU_SYSFS / "isolated"))
    run_cpus: JsonArray = list(others)
    job_cpus: JsonArray = list(cpus)
    idle: JsonArray = list(sorted(siblings - {core}))
    kept: JsonArray = list(isolated)
    return others, {"host": {"hostname": socket.gethostname(), "boot_id": boot},
                    "lsf_job": os.environ.get("LSB_JOBID"), "pass_thread_core": core,
                    "core_siblings_idle": idle, "run_cpus": run_cpus, "job_cpus": job_cpus,
                    "host_cpus": os.cpu_count(), "isolated_cpus": kept,
                    "core_isolated": core in isolated}

AffinityStatus leave_pass_cpu(int pass_cpu) noexcept {
  if (pass_cpu < 0) return AffinityStatus::inherited;
  if (pass_cpu >= CPU_SETSIZE) return AffinityStatus::failed;
  cpu_set_t set;
  CPU_ZERO(&set);
  if (pthread_getaffinity_np(pthread_self(), sizeof set, &set) != 0) return AffinityStatus::failed;
  if (CPU_ISSET(pass_cpu, &set)) {
    CPU_CLR(pass_cpu, &set);
    if (CPU_COUNT(&set) == 0) return AffinityStatus::shared_core;
    if (pthread_setaffinity_np(pthread_self(), sizeof set, &set) != 0) return AffinityStatus::failed;
  }
  cpu_set_t now;
  CPU_ZERO(&now);
  if (pthread_getaffinity_np(pthread_self(), sizeof now, &now) != 0) return AffinityStatus::failed;

            problems.append(f"the record's log left {d.get('passes_unlogged')} passes out")
        passes = tuple(_pass(p, i) for i, p in enumerate(_list(d.get("pass_entries"),
                                                                "record.pass_entries")))
        by_id = {p.pass_id: p for p in passes}
        if len(by_id) != len(passes):
            raise JudgeError("the record logs a pass twice")
        if any(not p.accounted for p in passes):
            problems.append("a logged pass carries no slot accounting")

# Commands of the one read-only login to CRB on 2026-11-11 at 21:00 UTC.
# Their output is probe-20261111T210017Z.txt in this directory.
# Run from the dev machine as:
#   ssh -o BatchMode=yes -o ConnectTimeout=10 jhan@192.168.2.45 'bash -s' < this script
# The first line is the RUNNING-marker check. The marker was absent, so the script went on
# (a present marker prints RUNNING_MARKER_PRESENT and exits with code 3; the saved output has no such line).
# No benchmark and no perf command is in this script.

if [ -e /tmp/positron-perf.RUNNING ]; then echo RUNNING_MARKER_PRESENT; exit 3; fi
echo "## host=$(hostname) utc=$(date -u +%FT%TZ) load=$(cut -d' ' -f1-3 /proc/loadavg) logins=$(who | wc -l)"
echo "## kernel=$(uname -r) bios=$(cat /sys/class/dmi/id/bios_version 2>/dev/null) bios_date=$(cat /sys/class/dmi/id/bios_date 2>/dev/null)"
echo "## sve_vl_bytes=$(cat /proc/sys/abi/sve_default_vector_length 2>/dev/null)"
echo "## gcc=$(gcc --version | head -1)"
echo "## march_native: $(gcc -march=native -Q --help=target 2>/dev/null | grep -E '^\s+-m(arch|cpu|tune)=' | tr -s ' \t' ' ' | tr '\n' ';')"
cd "$HOME/cache-perf/bench" 2>/dev/null || { echo "## NO_BENCH_DIR"; ls "$HOME" | head -20; exit 4; }
echo "## files:"; ls -la --time-style=long-iso bandwidth bandwidth.c Makefile 2>&1
echo "## sha256:"; sha256sum bandwidth.c bandwidth Makefile 2>&1
echo "## scmi_dev: $(ls /dev/scmi 2>&1 | tr '\n' ' ')"
echo "## cmn_events_total=$(ls /sys/bus/event_source/devices/arm_cmn_0/events 2>/dev/null | wc -l) dat2_or_rsp2: $(ls /sys/bus/event_source/devices/arm_cmn_0/events 2>/dev/null | grep -E 'dat2|rsp2|req2|snp2' | tr '\n' ' ')"
for f in k_read k_write k_copy k_asmread; do echo "=== $f ==="; objdump -d --no-show-raw-insn --disassemble=$f bandwidth | sed -n '/>:$/,$p'; done

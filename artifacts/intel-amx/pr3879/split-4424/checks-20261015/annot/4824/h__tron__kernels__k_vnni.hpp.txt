    1| +#pragma once
    2| +// The AVX-512 (the CPU's 512-bit vector instruction set) QK score reader of the
    3| +// K VNNI layout (h/tron/tensor/k_vnni.hpp). It serves partial pages, hosts
    4| +// without AMX (Advanced Matrix Extensions, the tile instruction set) and the
    5| +// AMX kill switch (TRON_AMX_DISABLE=1). No AMX instruction runs here. So any
    6| +// translation unit may include it. The layout itself, who writes it and who
    7| +// reads it are described in Note [K VNNI storage] (in kv_cache.hpp).
    8| +//
    9| +// Words used here: bf16 is the 16-bit brain floating-point format. fp16 is the
   10| +// 16-bit IEEE half format. fp32 is the 32-bit IEEE single format. VDPBF16PS is
   11| +// the AVX-512 instruction that multiplies bf16 pairs and adds them in fp32.
   12| +
   13| +#include <cstddef>
   14| +#include <cstdint>
   15| +#include <immintrin.h>
   16| +#include <type_traits>
   17| +
   18| +#include "common/attributes.hpp"
   19| +#include "common/numerics/bf16.hpp"
   20| +#include "common/numerics/fp16.hpp"
   21| +#include "tron/tensor/k_vnni.hpp"
   22| +
   23| +namespace tron::k_vnni {
   24| +
   25| +// Query heads scored per pass of qk_group: one accumulator register per head.
   26| +// A wider group (more query heads sharing one K/V head) is scored in passes of
   27| +// 16 heads.
   28| +inline constexpr size_t MAX_KV_MUL_16 = 16;
   29| +
   30| +// qk_group is a template. It is instantiated only for VNNI slots. Its
   31| +// qualified name stays declared for the discarded branch of
   32| +// self_attention.hpp's layout `if constexpr`.
   33| +
   34| +// Scores of one query group against the live tokens of one page, read from
   35| +// the VNNI plane with AVX-512. s[h * s_stride + t] = q_h . K_t (UNSCALED) for
   36| +// every token t whose bit is set in `live`. The other entries of s are left
   37| +// untouched (the caller keeps them at -inf). The K storage of a token whose
   38| +// bit is clear is never read (masked loads). So unwritten tokens cannot
   39| +// change the result. Every token block with a live bit must hold the VNNI
   40| +// layout (Note [Row-major tail block] in h/tron/tensor/k_vnni.hpp): the
   41| +// caller scores the tokens of a row-major block with the row-major dotter
   42| +// and keeps their bits clear here. Token lanes: one 64-byte load supplies
   43| +// one dimension pair of 16 tokens. The query's dimension pair is broadcast.
   44| +// VDPBF16PS accumulates 16 scores at once. So there is no horizontal
   45| +// reduction and no per-token gather. k_page is the page's packed K plane as
   46| +// its typed view (page::k_packed). This reader is one of the two
   47| +// layout-specific readers allowed to take the plane address. q points at
   48| +// kv_mul rows of 128 values (the group's heads, row stride 128). kv_mul is
   49| +// any group size: a group of more than 16 heads is scored 16 heads at a
   50| +// time. Every head's scores are its own accumulation. So the split changes
   51| +// no head's add order.
   52| +//   bf16 q: exact bf16 x bf16 products added in fp32 by VDPBF16PS. This is the
   53| +//     same instruction the row-major dotter<bf16, 128> uses. Only the order of
   54| +//     the fp32 adds differs (see Note [AMX attention dispatch]).
   55| +//   float q: the stored bf16 K values are widened to fp32 exactly (a shift).
   56| +//     They are then multiplied by the fp32 query with FMA (fused multiply-add),
   57| +//     as dotter<float, 128> does.
   58| +//   fp16 q: widened to fp32 once per call, then the float path.
   59| +template <size_t kv_mul, typename q_scalar>
   60| +TRON(inline)
   61| +void qk_group(k_vnni_view<const bf16, PAGE_TOKENS_64, HEAD_SIZE_128> k_page,
   62| +    const q_scalar* q,
   63| +    uint64_t live,
   64| +    float* s,
   65| +    size_t s_stride) noexcept {
   66| +  static_assert(kv_mul > 0, "a query group has at least one head");
   67| +  if constexpr (kv_mul > MAX_KV_MUL_16) {
   68| +    // The first 16 heads, then the rest (which may split again).
   69| +    qk_group<MAX_KV_MUL_16, q_scalar>(k_page, q, live, s, s_stride);
   70| +    qk_group<kv_mul - MAX_KV_MUL_16, q_scalar>(k_page,
   71| +        q + MAX_KV_MUL_16 * HEAD_SIZE_128, live, s + MAX_KV_MUL_16 * s_stride,
   72| +        s_stride);
   73| +  } else if constexpr (std::is_same_v<q_scalar, fp16>) {
   74| +    alignas(64) float qf[kv_mul * HEAD_SIZE_128];
   75| +    for (size_t i = 0; i < kv_mul * HEAD_SIZE_128; ++i) {
   76| +      qf[i] = float(q[i]);
   77| +    }
   78| +    qk_group<kv_mul, float>(k_page, qf, live, s, s_stride);
   79| +    return;
   80| +  } else {
   81| +    static_assert(std::is_same_v<q_scalar, bf16> || std::is_same_v<q_scalar, float>,
   82| +        "the VNNI-K reader serves bf16, float and fp16 queries");
   83| +    // Inside namespace tron::k_vnni the name `detail` is tron::k_vnni::detail. So
   84| +    // the access helper is named in full.
   85| +    const bf16* const plane = tron::detail::k_vnni_access::plane(k_page);
   86| +    for (size_t c = 0; c < TOKEN_BLOCKS_4; ++c) {
   87| +      const __mmask16 mask = __mmask16(live >> (c * BLOCK_TOKENS_16));
   88| +      if (mask == 0) continue;
   89| +      __m512 acc[kv_mul];
   90| +      for (size_t h = 0; h < kv_mul; ++h) {
   91| +        acc[h] = _mm512_setzero_ps();
   92| +      }
   93| +      for (size_t step = 0; step < DIM_STEPS_4; ++step) {
   94| +        const bf16* panel = plane + (step * TOKEN_BLOCKS_4 + c) * PANEL_ELEMS_512;
   95| +        for (size_t dp = 0; dp < STEP_PAIRS_16; ++dp) {
   96| +          // One dimension pair of the 16 tokens of this block. Lanes of dead tokens
   97| +          // are never loaded. Those lanes are zero.
   98| +          const __m512i k_pairs =
   99| +              _mm512_maskz_loadu_epi32(mask, panel + dp * ROW_ELEMS_32);
  100| +          const size_t d = step * STEP_DIMS_32 + PAIR_2 * dp;
  101| +          if constexpr (std::is_same_v<q_scalar, bf16>) {
  102| +            for (size_t h = 0; h < kv_mul; ++h) {
  103| +              uint32_t q_pair;
  104| +              std::memcpy(&q_pair, q + h * HEAD_SIZE_128 + d, sizeof(q_pair));
  105| +              acc[h] = _mm512_dpbf16_ps(
  106| +                  acc[h], (__m512bh)_mm512_set1_epi32(int(q_pair)), (__m512bh)k_pairs);
  107| +            }
  108| +          } else {
  109| +            // Even dimensions are in the low halves of the dwords, odd dimensions
  110| +            // in the high halves. Each becomes an exact fp32 with a shift or a mask.
  111| +            const __m512 k_even =
  112| +                _mm512_castsi512_ps(_mm512_slli_epi32(k_pairs, BF16_BITS_16));
  113| +            const __m512 k_odd = _mm512_castsi512_ps(_mm512_and_si512(
  114| +                k_pairs, _mm512_set1_epi32(int(HIGH_HALF_0XFFFF0000))));
  115| +            for (size_t h = 0; h < kv_mul; ++h) {
  116| +              acc[h] = _mm512_fmadd_ps(
  117| +                  _mm512_set1_ps(q[h * HEAD_SIZE_128 + d]), k_even, acc[h]);
  118| +              acc[h] = _mm512_fmadd_ps(
  119| +                  _mm512_set1_ps(q[h * HEAD_SIZE_128 + d + 1]), k_odd, acc[h]);
  120| +            }
  121| +          }
  122| +        }
  123| +      }
  124| +      for (size_t h = 0; h < kv_mul; ++h) {
  125| +        _mm512_mask_storeu_ps(s + h * s_stride + c * BLOCK_TOKENS_16, mask, acc[h]);
  126| +      }
  127| +    }
  128| +  }
  129| +}
  130| +
  131| +}  // namespace tron::k_vnni

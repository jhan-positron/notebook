    1| +// This file tests the VNNI (Vector Neural Network Instructions) K storage layout
    2| +// (Note [K VNNI storage] in kv_cache.hpp) through its typed owner and views
    3| +// (h/tron/tensor/k_vnni.hpp). The index map matches hand-computed anchors. The

   46| +// The bytes of one fp32 (the 32-bit IEEE single format). The bf16 (the 16-bit
   47| +// brain floating-point format) helpers move bit patterns through memcpy.
   48| +constexpr size_t F32_BYTES_4 = 4;
   49| +// Round-to-nearest-even (the nearest bf16 is chosen, a tie goes to the even one)
   50| +// bias added before the fp32 -> bf16 truncation.
   51| +constexpr uint32_t RNE_BIAS_0X7FFF = 0x7fffu;
   52| +// The largest bf16 bit pattern: the random-bits distributions cover all of them.
   53| +constexpr uint16_t BITS_MAX_0XFFFF = 0xffff;
   54| +// Poison for the plane slots no store may touch.
   55| +constexpr uint16_t POISON_0XA5A5 = 0xa5a5;
   56| +
   57| +uint16_t bf16_rne(float f) {
   58| +  uint32_t bits;
   59| +  std::memcpy(&bits, &f, F32_BYTES_4);
   60| +  bits += RNE_BIAS_0X7FFF + ((bits >> BF16_BITS_16) & 1u);
   61| +  return static_cast<uint16_t>(bits >> BF16_BITS_16);
   62| +}
   63| +
   64| +float bf16_to_float(uint16_t b) {
   65| +  uint32_t bits = static_cast<uint32_t>(b) << BF16_BITS_16;
   66| +  float f;
   67| +  std::memcpy(&f, &bits, F32_BYTES_4);
   68| +  return f;
   69| +}

   71| +using k_plane_t = tron::k_vnni_tensor<PAGE_TOKENS_64, HEAD_SIZE_128>;
   72| +// The plane's bytes copied out for the physical-layout checks. The owner is
   73| +// trivially copyable, so a memcpy of the owner is the plane's bytes.
   74| +using plane_bytes_t = std::array<tron::bf16, PAGE_TOKENS_64 * HEAD_SIZE_128>;
   75| +static_assert(sizeof(k_plane_t) == sizeof(plane_bytes_t),
   76| +    "the owner holds the plane and nothing else");
   77| +
   78| +struct alignas(64) plane_and_rows {
   79| +  k_plane_t plane;  // the packed K plane, written through the typed operations
   80| +  tron::bf16 rows[PAGE_TOKENS_64 * HEAD_SIZE_128];  // the same K, one row per token
   81| +};
   82| +
   83| +// One token's native row of `rows`, as the read-only typed source row.
   84| +tron::const_k_row_view<HEAD_SIZE_128> row_of(const plane_and_rows& buf, size_t tok) {
   85| +  return tron::k_source_row<HEAD_SIZE_128>(buf.rows + tok * HEAD_SIZE_128);
   86| +}
   87| +
   88| +// Every token's slots filled with one poison pattern, through 64 typed row
   89| +// stores (the owner has no raw access).
   90| +void poison_plane(k_plane_t& plane, uint16_t pattern) {
   91| +  alignas(64) tron::bf16 poison_row[HEAD_SIZE_128];
   92| +  for (auto& x : poison_row) {
   93| +    x = tron::bf16::from_bits(pattern);
   94| +  }
   95| +  for (size_t tok = 0; tok < PAGE_TOKENS_64; ++tok) {
   96| +    tron::store_row(
   97| +        plane.as_view(), tok, tron::k_source_row<HEAD_SIZE_128>(poison_row));
   98| +  }
   99| +}
  100| +
  101| +plane_bytes_t bytes_of(const k_plane_t& plane) {
  102| +  plane_bytes_t bytes;
  103| +  std::memcpy(bytes.data(), &plane, sizeof(plane));
  104| +  return bytes;
  105| +}

  109| +TEST_CASE("VNNI K index map matches hand-computed anchors", "[kv_data][k_vnni]") {
  110| +  // Positions derived by hand from the layout definition (Note [K VNNI
  111| +  // storage]). One panel = 512 elements = 1024 bytes (16 dimension-pair rows x 16
  112| +  // tokens x 2). One dimension step = 4 panels. One dimension-pair row = 32
  113| +  // elements. One token = 2 elements inside a row. The (17, 34) anchor is the
  114| +  // worked example of the 2026-09-04 single-K feasibility study (bf16 offset 2594).
  115| +  using tron::k_vnni::index;
  116| +  STATIC_REQUIRE(index(0, 0) == 0);
  117| +  STATIC_REQUIRE(index(0, 1) == 1);      // second value of the pair
  118| +  STATIC_REQUIRE(index(1, 0) == 2);      // next token: next 32-bit lane
  119| +  STATIC_REQUIRE(index(15, 0) == 30);    // last token of block 0
  120| +  STATIC_REQUIRE(index(16, 0) == 512);   // token block 1: next panel
  121| +  STATIC_REQUIRE(index(0, 2) == 32);     // dimension pair 1: next 64-byte row
  122| +  STATIC_REQUIRE(index(0, 31) == 481);   // last pair of step 0, second value
  123| +  STATIC_REQUIRE(index(0, 32) == 2048);  // dimension step 1: 4 panels later
  124| +  STATIC_REQUIRE(index(17, 34) == 2594);
  125| +  STATIC_REQUIRE(index(63, 127) == 8191);  // the last element of the plane
  126| +}

  128| +TEST_CASE(
  129| +    "VNNI K store and load are exact and token-independent", "[kv_data][k_vnni]") {
  130| +  constexpr uint32_t SEED_7 = 7;
  131| +  // The scrambled store order: token (i * 37 + 11) mod 64 for i = 0..63.
  132| +  constexpr size_t SCRAMBLE_STRIDE_37 = 37;
  133| +  constexpr size_t SCRAMBLE_OFFSET_11 = 11;
  134| +  auto buf = std::make_unique<plane_and_rows>();
  135| +  std::mt19937 gen(SEED_7);
  136| +  std::uniform_int_distribution<uint16_t> bits(0, BITS_MAX_0XFFFF);
  137| +  for (auto& x : buf->rows) {
  138| +    x = tron::bf16::from_bits(bits(gen));
  139| +  }
  140| +  // Poison: a wrong index would leave poison where a value belongs, or put
  141| +  // a value where poison belongs.
  142| +  poison_plane(buf->plane, POISON_0XA5A5);
  143| +  // Store every token in a scrambled order (tokens own disjoint slots).
  144| +  for (size_t i = 0; i < PAGE_TOKENS_64; ++i) {
  145| +    const size_t tok = (i * SCRAMBLE_STRIDE_37 + SCRAMBLE_OFFSET_11) % PAGE_TOKENS_64;
  146| +    tron::store_row(buf->plane.as_view(), tok, row_of(*buf, tok));
  147| +  }
  148| +  SECTION("the plane holds every value at the documented position") {
  149| +    const plane_bytes_t bytes = bytes_of(buf->plane);
  150| +    for (size_t tok = 0; tok < PAGE_TOKENS_64; ++tok) {
  151| +      for (size_t d = 0; d < HEAD_SIZE_128; ++d) {
  152| +        // store_row addresses the plane through its own pair-row offsets, not
  153| +        // through index(). Agreement here ties the store to the index map.
  154| +        const size_t at = tron::k_vnni::index(tok, d);
  155| +        REQUIRE(bytes[at].to_bits() == buf->rows[tok * HEAD_SIZE_128 + d].to_bits());
  156| +      }
  157| +    }
  158| +  }
  159| +  SECTION("at() reads the plane by (token, dim)") {
  160| +    auto const plane = std::as_const(buf->plane).as_view();
  161| +    for (size_t tok = 0; tok < PAGE_TOKENS_64; ++tok) {
  162| +      for (size_t d = 0; d < HEAD_SIZE_128; ++d) {
  163| +        REQUIRE(
  164| +            plane.at(tok, d).to_bits() == buf->rows[tok * HEAD_SIZE_128 + d].to_bits());
  165| +      }
  166| +    }
  167| +  }
  168| +  SECTION("load_row inverts store_row for every token") {
  169| +    alignas(64) tron::bf16 row[HEAD_SIZE_128];
  170| +    for (size_t tok = 0; tok < PAGE_TOKENS_64; ++tok) {
  171| +      tron::load_row(std::as_const(buf->plane).as_view(), tok,
  172| +          tron::k_destination_row<HEAD_SIZE_128>(row));
  173| +      REQUIRE(std::memcmp(row, buf->rows + tok * HEAD_SIZE_128, sizeof(row)) == 0);
  174| +    }
  175| +  }
  176| +  SECTION("rewriting one token changes no other token's slots") {
  177| +    constexpr size_t REWRITTEN_TOKEN_29 = 29;
  178| +    for (size_t d = 0; d < HEAD_SIZE_128; ++d) {
  179| +      buf->rows[REWRITTEN_TOKEN_29 * HEAD_SIZE_128 + d] =
  180| +          tron::bf16::from_bits(bits(gen));
  181| +    }
  182| +    tron::store_row(
  183| +        buf->plane.as_view(), REWRITTEN_TOKEN_29, row_of(*buf, REWRITTEN_TOKEN_29));
  184| +    const plane_bytes_t bytes = bytes_of(buf->plane);
  185| +    for (size_t t = 0; t < PAGE_TOKENS_64; ++t) {
  186| +      for (size_t d = 0; d < HEAD_SIZE_128; ++d) {
  187| +        REQUIRE(bytes[tron::k_vnni::index(t, d)].to_bits() ==
  188| +                buf->rows[t * HEAD_SIZE_128 + d].to_bits());
  189| +      }
  190| +    }
  191| +  }
  192| +  SECTION("copy_token moves one token between planes, other tokens untouched") {
  193| +    constexpr size_t SOURCE_TOKEN_5 = 5;
  194| +    constexpr size_t DESTINATION_TOKEN_42 = 42;
  195| +    auto other = std::make_unique<plane_and_rows>();
  196| +    poison_plane(other->plane, POISON_0XA5A5);
  197| +    tron::copy_token(other->plane.as_view(), DESTINATION_TOKEN_42,

  212| +TEST_CASE("VNNI K block save equals the per-token row store, bit for bit",
  213| +    "[kv_data][k_vnni]") {
  214| +  // store_block (16 tokens transposed in registers, whole-line stores) is
  215| +  // the prefill store of save_k (Note [Shared block save] in model.hpp). It
  216| +  // must leave the plane exactly as store_row per present token does. It
  217| +  // must not touch the slots of absent tokens (poison stays poison). The
  218| +  // masks cover: full block, single tokens, halves, alternating lanes, the
  219| +  // empty block, and random subsets.
  220| +  constexpr uint32_t SEED_19 = 19;
  221| +  constexpr unsigned TOKEN_0_0X0001 = 0x0001;
  222| +  constexpr unsigned TOKEN_15_0X8000 = 0x8000;
  223| +  constexpr unsigned LOW_HALF_0X00FF = 0x00ff;
  224| +  constexpr unsigned HIGH_HALF_0XFF00 = 0xff00;
  225| +  constexpr unsigned EVEN_TOKENS_0X5555 = 0x5555;
  226| +  constexpr unsigned ODD_TOKENS_0XAAAA = 0xaaaa;
  227| +  constexpr unsigned EMPTY_BLOCK_0X0000 = 0x0000;
  228| +  constexpr unsigned MIXED_0X1234 = 0x1234;
  229| +  constexpr unsigned ALL_BUT_TOKEN_0_0XFFFE = 0xfffe;
  230| +  constexpr unsigned ALL_BUT_TOKEN_15_0X7FFF = 0x7fff;
  231| +  constexpr unsigned NIBBLES_0X0F0F = 0x0f0f;
  232| +  constexpr unsigned TOKEN_1_0X0002 = 0x0002;
  233| +  constexpr unsigned TOKEN_14_0X4000 = 0x4000;
  234| +  constexpr int N_RANDOM_MASKS_24 = 24;
  235| +  auto ref = std::make_unique<plane_and_rows>();
  236| +  auto blk = std::make_unique<plane_and_rows>();
  237| +  std::mt19937 gen(SEED_19);
  238| +  std::uniform_int_distribution<uint16_t> bits(0, BITS_MAX_0XFFFF);
  239| +  for (auto& x : ref->rows) {
  240| +    x = tron::bf16::from_bits(bits(gen));
  241| +  }
  242| +  std::vector<unsigned> masks = {FULL_BLOCK_0XFFFF, TOKEN_0_0X0001, TOKEN_15_0X8000,
  243| +      LOW_HALF_0X00FF, HIGH_HALF_0XFF00, EVEN_TOKENS_0X5555, ODD_TOKENS_0XAAAA,
  244| +      EMPTY_BLOCK_0X0000, MIXED_0X1234, ALL_BUT_TOKEN_0_0XFFFE, ALL_BUT_TOKEN_15_0X7FFF,
  245| +      NIBBLES_0X0F0F, TOKEN_1_0X0002, TOKEN_14_0X4000};
  246| +  for (int i = 0; i < N_RANDOM_MASKS_24; ++i) {
  247| +    masks.push_back(bits(gen));
  248| +  }
  249| +  size_t checked = 0;
  250| +  for (size_t c = 0; c < TOKEN_BLOCKS_4; ++c) {
  251| +    for (unsigned m : masks) {
  252| +      poison_plane(ref->plane, POISON_0XA5A5);
  253| +      blk->plane = ref->plane;  // the owner copy: same bytes
  254| +      const tron::bf16* rows[BLOCK_TOKENS_16] = {};
  255| +      for (size_t t = 0; t < BLOCK_TOKENS_16; ++t) {
  256| +        if (!((m >> t) & 1u)) continue;
  257| +        const size_t tok = c * BLOCK_TOKENS_16 + t;
  258| +        rows[t] = ref->rows + tok * HEAD_SIZE_128;
  259| +        tron::store_row(ref->plane.as_view(), tok, row_of(*ref, tok));
  260| +      }
  261| +      tron::store_block(blk->plane.as_view(), c, __mmask16(m), rows);
  262| +      REQUIRE(std::memcmp(&blk->plane, &ref->plane, sizeof(ref->plane)) == 0);
  263| +      ++checked;
  264| +    }
  265| +  }
  266| +  REQUIRE(checked == TOKEN_BLOCKS_4 * masks.size());
  267| +  SECTION("a block save on top of other blocks changes only its own block") {
  268| +    constexpr size_t NEW_BLOCK_2 = 2;
  269| +    constexpr size_t NEW_FIRST_TOKEN_32 = NEW_BLOCK_2 * BLOCK_TOKENS_16;
  270| +    constexpr size_t NEW_END_TOKEN_48 = NEW_FIRST_TOKEN_32 + BLOCK_TOKENS_16;
  271| +    static_assert(NEW_FIRST_TOKEN_32 == 32 && NEW_END_TOKEN_48 == 48,
  272| +        "every constant's name states its value");
  273| +    for (size_t tok = 0; tok < PAGE_TOKENS_64; ++tok) {
  274| +      tron::store_row(ref->plane.as_view(), tok, row_of(*ref, tok));
  275| +    }
  276| +    blk->plane = ref->plane;
  277| +    // New rows for block 2 only.
  278| +    for (size_t tok = NEW_FIRST_TOKEN_32; tok < NEW_END_TOKEN_48; ++tok) {
  279| +      for (size_t d = 0; d < HEAD_SIZE_128; ++d) {
  280| +        ref->rows[tok * HEAD_SIZE_128 + d] = tron::bf16::from_bits(bits(gen));
  281| +      }
  282| +    }
  283| +    const tron::bf16* rows[BLOCK_TOKENS_16];
  284| +    for (size_t t = 0; t < BLOCK_TOKENS_16; ++t) {
  285| +      rows[t] = ref->rows + (NEW_FIRST_TOKEN_32 + t) * HEAD_SIZE_128;
  286| +      tron::store_row(ref->plane.as_view(), NEW_FIRST_TOKEN_32 + t,
  287| +          row_of(*ref, NEW_FIRST_TOKEN_32 + t));
  288| +    }
  289| +    tron::store_block(blk->plane.as_view(), NEW_BLOCK_2, FULL_BLOCK_0XFFFF, rows);
  290| +    REQUIRE(std::memcmp(&blk->plane, &ref->plane, sizeof(ref->plane)) == 0);
  291| +    alignas(64) tron::bf16 row[HEAD_SIZE_128];
  292| +    for (size_t tok = 0; tok < PAGE_TOKENS_64; ++tok) {
  293| +      tron::load_row(std::as_const(blk->plane).as_view(), tok,
  294| +          tron::k_destination_row<HEAD_SIZE_128>(row));
  295| +      REQUIRE(std::memcmp(row, ref->rows + tok * HEAD_SIZE_128, sizeof(row)) == 0);
  296| +    }
  297| +  }

  299| +TEST_CASE("row-major tail block converts to the VNNI layout and back, bit for bit",
  300| +    "[kv_data][k_vnni]") {
  301| +  // Note [Row-major tail block] in h/tron/tensor/k_vnni.hpp: before its 16th
  302| +  // row is saved a block holds contiguous 256-byte rows (row_major_row) inside
  303| +  // its own 4 panels. block_to_vnni must produce the same bytes as store_row
  304| +  // per token, touch no other block, and block_to_rows must invert it.
  305| +  constexpr uint32_t SEED_23 = 23;
  306| +  constexpr size_t ROW_BYTES_256 = HEAD_SIZE_128 * sizeof(tron::bf16);
  307| +  constexpr size_t PANEL_BYTES_1024 = tron::k_vnni::PANEL_BYTES_1024;
  308| +  static_assert(ROW_BYTES_256 == 256, "the constant's name states its value");
  309| +  auto ref = std::make_unique<plane_and_rows>();
  310| +  auto blk = std::make_unique<plane_and_rows>();
  311| +  std::mt19937 gen(SEED_23);
  312| +  std::uniform_int_distribution<uint16_t> bits(0, BITS_MAX_0XFFFF);
  313| +  for (auto& x : ref->rows) {
  314| +    x = tron::bf16::from_bits(bits(gen));
  315| +  }
  316| +  SECTION("row_major_row rows are distinct, 64-byte aligned, and inside the block's "
  317| +          "panels") {
  318| +    // The owner is the plane (static_assert above): offsets are taken from it.
  319| +    const char* base = reinterpret_cast<const char*>(&blk->plane);
  320| +    const auto plane = std::as_const(blk->plane).as_view();
  321| +    for (size_t tok = 0; tok < PAGE_TOKENS_64; ++tok) {
  322| +      const size_t c = tok / BLOCK_TOKENS_16;
  323| +      const char* row =
  324| +          reinterpret_cast<const char*>(tron::row_major_row(plane, tok).data);
  325| +      const size_t off = static_cast<size_t>(row - base);
  326| +      CAPTURE(tok, off);
  327| +      REQUIRE(off % tron::k_vnni::LINE_BYTES_64 == 0);
  328| +      // Inside panel (s, c) for some dimension step s: those are the block's bytes.
  329| +      const size_t panel = off / PANEL_BYTES_1024;
  330| +      REQUIRE(panel % TOKEN_BLOCKS_4 == c);
  331| +      REQUIRE(off + ROW_BYTES_256 <= panel * PANEL_BYTES_1024 + PANEL_BYTES_1024);
  332| +      for (size_t other = 0; other < tok; ++other) {
  333| +        const char* orow =
  334| +            reinterpret_cast<const char*>(tron::row_major_row(plane, other).data);
  335| +        REQUIRE(static_cast<size_t>(std::abs(orow - row)) >= ROW_BYTES_256);
  336| +      }
  337| +    }
  338| +  }
  339| +  SECTION("block_to_vnni equals store_row per token and touches no other block") {
  340| +    for (size_t c = 0; c < TOKEN_BLOCKS_4; ++c) {
  341| +      poison_plane(ref->plane, POISON_0XA5A5);
  342| +      blk->plane = ref->plane;
  343| +      for (size_t t = 0; t < BLOCK_TOKENS_16; ++t) {
  344| +        const size_t tok = c * BLOCK_TOKENS_16 + t;
  345| +        tron::store_row(ref->plane.as_view(), tok, row_of(*ref, tok));
  346| +        tron::store_row_major(blk->plane.as_view(), tok, row_of(*ref, tok));
  347| +      }
  348| +      tron::block_to_vnni(blk->plane.as_view(), c);
  349| +      CAPTURE(c);
  350| +      REQUIRE(std::memcmp(&blk->plane, &ref->plane, sizeof(ref->plane)) == 0);
  351| +      tron::block_to_rows(blk->plane.as_view(), c);
  352| +      const auto plane = std::as_const(blk->plane).as_view();
  353| +      for (size_t t = 0; t < BLOCK_TOKENS_16; ++t) {
  354| +        const size_t tok = c * BLOCK_TOKENS_16 + t;
  355| +        CAPTURE(tok);
  356| +        REQUIRE(std::memcmp(tron::row_major_row(plane, tok).data,
  357| +                    ref->rows + tok * HEAD_SIZE_128, ROW_BYTES_256) == 0);
  358| +      }
  359| +      // The other blocks' bytes are still poison after both conversions.
  360| +      for (size_t tok = 0; tok < PAGE_TOKENS_64; ++tok) {
  361| +        if (tok / BLOCK_TOKENS_16 == c) continue;
  362| +        alignas(tron::k_vnni::LINE_BYTES_64) tron::bf16 row[HEAD_SIZE_128];
  363| +        tron::load_row(plane, tok, tron::k_destination_row<HEAD_SIZE_128>(row));
  364| +        for (size_t d = 0; d < HEAD_SIZE_128; ++d) {
  365| +          REQUIRE(row[d].to_bits() == POISON_0XA5A5);
  366| +        }
  367| +      }
  368| +    }

  374| +// The reader against the row-major dotter for one query-group size (KV_MUL
  375| +// query heads sharing the K/V head). qk_group scores groups wider than 16 heads
  376| +// in passes of 16. 20 heads is an uneven split (16 + 4). 32 heads is an even
  377| +// split (16 + 16).
  378| +template <size_t KV_MUL>
  379| +void check_reader_against_dotter() {
  380| +  // Same numerics contract as the AMX kernels (Note [AMX attention dispatch]).
  381| +  // For the bf16 query both readers add the same exact bf16 x bf16 products in
  382| +  // fp32. For the fp32 query both widen K to fp32 and use FMA (fused
  383| +  // multiply-add). So only the add order may differ. An fp16 (16-bit IEEE
  384| +  // half) query is widened to fp32 once by the reader. So it is checked
  385| +  // against the dotter on the widened query. Envelope: 1e-4 of the
  386| +  // absolute-product mass plus 1e-6. Tokens outside the live mask keep their
  387| +  // preset value. Their storage holds NaN (not a number) poison. A stray load
  388| +  // would spread it into the scores.
  389| +  constexpr uint32_t SEED_31 = 31;
  390| +  // Queries and keys are drawn from [-1, 1).
  391| +  constexpr float VALUE_RANGE_1 = 1.f;
  392| +  // Masks with holes: two mt19937 outputs (32 bits each) make one 64-token mask.
  393| +  constexpr int N_HOLE_MASKS_24 = 24;
  394| +  constexpr int GEN_WORD_BITS_32 = 32;
  395| +  // Single live tokens: the last token, the first token of token block 1.
  396| +  constexpr size_t LAST_TOKEN_63 = 63;
  397| +  constexpr size_t BLOCK_1_FIRST_TOKEN_16 = 16;
  398| +  // A quiet NaN (the NaN kind that raises no exception): the storage of dead tokens.
  399| +  constexpr uint16_t NAN_POISON_0X7FC1 = 0x7fc1;
  400| +  // Dead tokens keep this preset. A stray load would overwrite it.
  401| +  constexpr float PRESET_NEG_INF = -std::numeric_limits<float>::infinity();
  402| +  constexpr double REL_TOL_1E_MINUS_4 = 1e-4;
  403| +  constexpr double ABS_TOL_1E_MINUS_6 = 1e-6;
  404| +  auto buf = std::make_unique<plane_and_rows>();
  405| +  std::mt19937 gen(SEED_31);
  406| +  std::uniform_real_distribution<float> u(-VALUE_RANGE_1, VALUE_RANGE_1);
  407| +  alignas(64) tron::bf16 q_bf16[KV_MUL * HEAD_SIZE_128];
  408| +  alignas(64) float q_f32[KV_MUL * HEAD_SIZE_128];
  409| +  alignas(64) tron::fp16 q_f16[KV_MUL * HEAD_SIZE_128];
  410| +  alignas(64) float q_f16_widened[KV_MUL * HEAD_SIZE_128];  // the fp16 values as fp32
  411| +  for (size_t i = 0; i < KV_MUL * HEAD_SIZE_128; ++i) {
  412| +    q_bf16[i] = tron::bf16::from_bits(bf16_rne(u(gen)));
  413| +    q_f32[i] = u(gen);
  414| +    q_f16[i] = tron::fp16(u(gen));
  415| +    q_f16_widened[i] = float(q_f16[i]);
  416| +  }
  417| +  for (auto& x : buf->rows) {
  418| +    x = tron::bf16::from_bits(bf16_rne(u(gen)));
  419| +  }
  420| +  std::vector<uint64_t> masks;
  421| +  for (size_t n = 0; n <= PAGE_TOKENS_64; ++n) {  // every prefix, 0 and 64 included
  422| +    masks.push_back(n == PAGE_TOKENS_64 ? ~uint64_t{0} : ((uint64_t{1} << n) - 1));
  423| +  }
  424| +  for (int i = 0; i < N_HOLE_MASKS_24; ++i) {  // holes
  425| +    masks.push_back((uint64_t(gen()) << GEN_WORD_BITS_32) | uint64_t(gen()));
  426| +  }
  427| +  masks.push_back(uint64_t{1} << LAST_TOKEN_63);
  428| +  masks.push_back(uint64_t{1} << BLOCK_1_FIRST_TOKEN_16);
  429| +
  430| +  for (uint64_t live : masks) {
  431| +    // Only live tokens are stored. The rest of the plane is NaN poison.
  432| +    poison_plane(buf->plane, NAN_POISON_0X7FC1);
  433| +    for (size_t tok = 0; tok < PAGE_TOKENS_64; ++tok) {
  434| +      if ((live >> tok) & 1) {
  435| +        tron::store_row(buf->plane.as_view(), tok, row_of(*buf, tok));
  436| +      }
  437| +    }
  438| +    alignas(64) float s_bf16[KV_MUL][PAGE_TOKENS_64];
  439| +    alignas(64) float s_f32[KV_MUL][PAGE_TOKENS_64];
  440| +    alignas(64) float s_f16[KV_MUL][PAGE_TOKENS_64];
  441| +    for (auto* scores : {&s_bf16, &s_f32, &s_f16}) {
  442| +      for (auto& row : *scores) {
  443| +        for (auto& x : row) {
  444| +          x = PRESET_NEG_INF;
  445| +        }
  446| +      }
  447| +    }
  448| +    auto const plane = std::as_const(buf->plane).as_view();
  449| +    tron::k_vnni::qk_group<KV_MUL>(plane, q_bf16, live, &s_bf16[0][0], PAGE_TOKENS_64);
  450| +    tron::k_vnni::qk_group<KV_MUL>(plane, q_f32, live, &s_f32[0][0], PAGE_TOKENS_64);
  451| +    tron::k_vnni::qk_group<KV_MUL>(plane, q_f16, live, &s_f16[0][0], PAGE_TOKENS_64);
  452| +    for (size_t h = 0; h < KV_MUL; ++h) {
  453| +      tron::const_view<tron::bf16, tron::VIEW_ALIGNED_TRUE, tron::VIEW_DMA_FALSE,
  454| +          tron::seq<HEAD_SIZE_128>>
  455| +          qv_bf16{q_bf16 + h * HEAD_SIZE_128};
  456| +      tron::const_view<float, tron::VIEW_ALIGNED_TRUE, tron::VIEW_DMA_FALSE,
  457| +          tron::seq<HEAD_SIZE_128>>
  458| +          qv_f32{q_f32 + h * HEAD_SIZE_128};
  459| +      tron::const_view<float, tron::VIEW_ALIGNED_TRUE, tron::VIEW_DMA_FALSE,
  460| +          tron::seq<HEAD_SIZE_128>>
  461| +          qv_f16_widened{q_f16_widened + h * HEAD_SIZE_128};
  462| +      tron::dotter<tron::bf16, HEAD_SIZE_128> dot_bf16(qv_bf16);
  463| +      tron::dotter<float, HEAD_SIZE_128> dot_f32(qv_f32);
  464| +      tron::dotter<float, HEAD_SIZE_128> dot_f16_widened(qv_f16_widened);
  465| +      for (size_t t = 0; t < PAGE_TOKENS_64; ++t) {
  466| +        CAPTURE(live, h, t);
  467| +        if (!((live >> t) & 1)) {
  468| +          REQUIRE(s_bf16[h][t] == PRESET_NEG_INF);
  469| +          REQUIRE(s_f32[h][t] == PRESET_NEG_INF);
  470| +          REQUIRE(s_f16[h][t] == PRESET_NEG_INF);
  471| +          continue;
  472| +        }
  473| +        tron::const_view<tron::bf16, tron::VIEW_ALIGNED_TRUE, tron::VIEW_DMA_FALSE,
  474| +            tron::seq<HEAD_SIZE_128>>
  475| +            kv{buf->rows + t * HEAD_SIZE_128};
  476| +        double mass_bf16 = 0, mass_f32 = 0, mass_f16 = 0;
  477| +        for (size_t d = 0; d < HEAD_SIZE_128; ++d) {
  478| +          const double k = bf16_to_float(buf->rows[t * HEAD_SIZE_128 + d].to_bits());
  479| +          mass_bf16 += std::fabs(
  480| +              double(bf16_to_float(q_bf16[h * HEAD_SIZE_128 + d].to_bits())) * k);
  481| +          mass_f32 += std::fabs(double(q_f32[h * HEAD_SIZE_128 + d]) * k);
  482| +          mass_f16 += std::fabs(double(q_f16_widened[h * HEAD_SIZE_128 + d]) * k);
  483| +        }
  484| +        REQUIRE(std::fabs(double(s_bf16[h][t]) - double(dot_bf16(kv))) <=
  485| +                REL_TOL_1E_MINUS_4 * mass_bf16 + ABS_TOL_1E_MINUS_6);
  486| +        REQUIRE(std::fabs(double(s_f32[h][t]) - double(dot_f32(kv))) <=
  487| +                REL_TOL_1E_MINUS_4 * mass_f32 + ABS_TOL_1E_MINUS_6);
  488| +        REQUIRE(std::fabs(double(s_f16[h][t]) - double(dot_f16_widened(kv))) <=
  489| +                REL_TOL_1E_MINUS_4 * mass_f16 + ABS_TOL_1E_MINUS_6);
  490| +      }
  491| +    }
  492| +  }
  493| +}
  494| +

  514| +
  515| +// One slot, one KV head of 128 dimensions: a VNNI slot under TRON_K_VNNI, a
  516| +// row-major slot otherwise. The accessors and the append path below are
  517| +// layout-agnostic. So the same checks hold in both builds.
  518| +constexpr size_t N_SLOTS_1 = 1;
  519| +constexpr size_t N_KV_HEADS_1 = 1;
  520| +using book_t = tron::
  521| +    uniform_kv_cache<N_SLOTS_1, tron::kv_geometry{N_KV_HEADS_1, HEAD_SIZE_128}, 0>;
  522| +using page_t = book_t::page_t;
  523| +constexpr tron::kv_slot_id SLOT_0(0);
  524| +// Every book below has one page.
  525| +constexpr size_t N_PAGES_1 = 1;
  526| +
  527| +// Distinct finite bf16 bit patterns per (salt, token, dimension). The salt selects
  528| +// one of four ranges of 16 Ki (0x4000) bit patterns. token * 128 + d (< 0x2000)
  529| +// selects the position inside it. Every pattern is shifted up by 0x0800
  530| +// (PATTERN_BASE_0X0800). So every (token, dimension) of one salt is distinct.
  531| +// No pattern reaches the inf (infinity) / NaN range 0x7f80..0x7fff or
  532| +// 0xff80..0xffff. NaN is unequal to itself. That would hide a correct copy.
  533| +void fill_row(std::array<tron::bf16, HEAD_SIZE_128>& row, size_t token, size_t salt) {
  534| +  constexpr size_t N_SALT_RANGES_4 = 4;
  535| +  constexpr size_t SALT_RANGE_0X4000 = 0x4000;
  536| +  constexpr size_t PATTERN_BASE_0X0800 = 0x0800;
  537| +  for (size_t d = 0; d < HEAD_SIZE_128; ++d) {
  538| +    row[d] = tron::bf16::from_bits(static_cast<uint16_t>(
  539| +        PATTERN_BASE_0X0800 + (salt % N_SALT_RANGES_4) * SALT_RANGE_0X4000 +
  540| +        token * HEAD_SIZE_128 + d));
  541| +  }
  542| +}
  543| +
  544| +// Bit-exact comparison of two rows. bf16 operator== compares values. The test
  545| +// must see every stored bit.
  546| +bool same_bits(const std::array<tron::bf16, HEAD_SIZE_128>& a,
  547| +    const std::array<tron::bf16, HEAD_SIZE_128>& b) {
  548| +  for (size_t d = 0; d < HEAD_SIZE_128; ++d) {
  549| +    if (a[d].to_bits() != b[d].to_bits()) return false;
  550| +  }
  551| +  return true;
  552| +}
